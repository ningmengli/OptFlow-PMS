```sql
-- API-034 visit/transfer 9 步事务(235B §4 冻结)
-- 必须在 1 个 DB 事务内,任一失败 ROLLBACK
-- 前置: 旧 visitId, 转诊原因 transferReason

START TRANSACTION;

-- 步骤 1:验证旧 Visit = IN_CONSULTATION
SELECT status FROM visit WHERE visitId = ? FOR UPDATE;
-- 若 status != 'IN_CONSULTATION' → ROLLBACK + 错误

-- 步骤 2:创建新 Visit(DRAFT, 转诊来源)
INSERT INTO visit (
  companyId, appointmentId, patientId, employeeId, consultRoomId,
  visitType, status, transferredFromVisitId, transferredFromEmployeeId, transferredAt
) VALUES (
  ?, ?, ?, ?, ?,
  'NORMAL', 'DRAFT', ?, ?, NOW(3)
);
-- 获得 newVisitId

-- 步骤 3:创建新 TriageQueue(IN_POOL, employeeId/consultRoomId NULL)
INSERT INTO triage_queue (
  companyId, visitId, appointmentId, consultRoomId, employeeId, status
) VALUES (
  ?, ?, ?, NULL, NULL, 'IN_POOL'
);

-- 步骤 4:更新 Appointment.currentVisitId
UPDATE appointment SET currentVisitId = ? WHERE appointmentId = ?;

-- 步骤 5:更新 Appointment.status
UPDATE appointment SET status = 'TRIAGE_WAITING' WHERE appointmentId = ?;

-- 步骤 6:旧 Visit 终态
UPDATE visit SET status = 'TRANSFERRED', transferredAt = NOW(3) WHERE visitId = ?;

-- 步骤 7:旧 TriageQueue 终态
UPDATE triage_queue SET status = 'REMOVED' WHERE visitId = ? AND status != 'REMOVED';

-- 步骤 8:旧 ReceptionQueue 终态
UPDATE reception_queue SET status = 'DONE', cancelReason = 'TRANSFERRED'
WHERE visitId = ? AND status != 'DONE';

-- 步骤 9:COMMIT
COMMIT;
```
